% ==============================================================================
% File Name: latex/section_standalone.tex
%
% Description:
%   Minimal standalone XeLaTeX wrapper template for compiling an individual section
%   into a single-page PDF. Dynamically loads the target section using the TeX
%   \jobname macro (\input{sections/\jobname}).
%
% How to Use:
%   Compiled programmatically with latexmk and a specified jobname:
%       latexmk -xelatex -jobname=<section_name> latex/section_standalone.tex
%   (e.g., latexmk -jobname=experience → compiles sections/experience.tex to experience.pdf).
%
% Where It Is Used:
%   - Consumed by scripts/compile_sections_pdf.sh to generate individual PDFs
%     in output/pdf/sections/<section_name>.pdf.
% ==============================================================================

\documentclass[11pt, a4paper]{article}

\usepackage{styles/resume}

\begin{document}
\pagestyle{empty}

% Route to the correct section file via \jobname
\edef\sectionfile{sections/\jobname}
\input{\sectionfile}

\end{document}
